```javascript
import express from 'express';
import request from 'supertest';
import { jest } from '@jest/globals';

jest.mock('../dataset/external/iampranavdhar__Library-Management-System-MERN/backend/models/User.js');

import User from '../dataset/external/iampranavdhar__Library-Management-System-MERN/backend/models/User.js';
import router from '../dataset/external/iampranavdhar__Library-Management-System-MERN/backend/routes/users.js';

const app = express();
app.use(express.json());
app.use('/', router);

describe('Users Routes', () => {
    beforeEach(() => {
        jest.clearAllMocks();
    });

    describe('GET /getuser/:id', () => {
        it('should get user by id and exclude password and updatedAt', async () => {
            const mockUserDoc = {
                _doc: {
                    _id: 'user123',
                    username: 'john_doe',
                    email: 'john@example.com',
                    password: 'hashedpassword',
                    updatedAt: '2023-01-01'
                }
            };

            const populateSecond = jest.fn().mockResolvedValue(mockUserDoc);
            const populateFirst = jest.fn().mockReturnValue({ populate: populateSecond });
            User.findById.mockReturnValue({ populate: populateFirst });

            const res = await request(app).get('/getuser/user123');

            expect(res.status).toBe(200);
            expect(res.body).toEqual({
                _id: 'user123',
                username: 'john_doe',
                email: 'john@example.com'
            });
            expect(User.findById).toHaveBeenCalledWith('user123');
            expect(populateFirst).toHaveBeenCalledWith('activeTransactions');
            expect(populateSecond).toHaveBeenCalledWith('prevTransactions');
        });

        it('should return 500 if DB error occurs', async () => {
            User.findById.mockImplementation(() => {
                throw new Error('Database Error');
            });

            const res = await request(app).get('/getuser/user123');

            expect(res.status).toBe(500);
        });
    });

    describe('GET /allmembers', () => {
        it('should return all members sorted by _id descending', async () => {
            const mockUsers = [
                { _id: '2', username: 'user2' },
                { _id: '1', username: 'user1' }
            ];

            const sortMock = jest.fn().mockResolvedValue(mockUsers);
            const populateSecond = jest.fn().mockReturnValue({ sort: sortMock });
            const populateFirst = jest.fn().mockReturnValue({ populate: populateSecond });
            User.find.mockReturnValue({ populate: populateFirst });

            const res = await request(app).get('/allmembers');

            expect(res.status).toBe(200);
            expect(res.body).toEqual(mockUsers);
            expect(User.find).toHaveBeenCalledWith({});
            expect(populateFirst).toHaveBeenCalledWith('activeTransactions');
            expect(populateSecond).toHaveBeenCalledWith('prevTransactions');
            expect(sortMock).toHaveBeenCalledWith({ _id: -1 });
        });

        it('should return 500 if DB error occurs', async () => {
            User.find.mockImplementation(() => {
                throw new Error('Database Error');
            });

            const res = await request(app).get('/allmembers');

            expect(res.status).toBe(500);
        });
    });

    describe('PUT /updateuser/:id', () => {
        it('should return 403 if user is not authorized to update account', async () => {
            const res = await request(app)
                .put('/updateuser/user123')
                .send({ userId: 'otherUser', isAdmin: false });

            expect(res.status).toBe(403);
            expect(res.body).toBe('You can update only your account!');
        });

        it('should update user successfully without password change', async () => {
            User.findByIdAndUpdate.mockResolvedValue({});

            const res = await request(app)
                .put('/updateuser/user123')
                .send({ userId: 'user123', username: 'newname' });

            expect(res.status).toBe(200);
            expect(res.body).toBe('Account has been updated');
            expect(User.findByIdAndUpdate).toHaveBeenCalledWith('user123', {
                $set: { userId: 'user123', username: 'newname' }
            });
        });

        it('should return 500 if password hashing or update fails', async () => {
            const res = await request(app)
                .put('/updateuser/user123')
                .send({ userId: 'user123', password: 'newpassword' });

            expect(res.status).toBe(500);
        });

        it('should return 500 if findByIdAndUpdate throws an error', async () => {
            User.findByIdAndUpdate.mockRejectedValue(new Error('Update error'));

            const res = await request(app)
                .put('/updateuser/user123')
                .send({ userId: 'user123', username: 'newname' });

            expect(res.status).toBe(500);
        });
    });

    describe('PUT /:id/move-to-activetransactions', () => {
        it('should return 403 if request is not made by admin', async () => {
            const res = await request(app)
                .put('/trans123/move-to-activetransactions')
                .send({ isAdmin: false, userId: 'user123' });

            expect(res.status).toBe(403);
            expect(res.body).toBe('Only Admin can add a transaction');
        });

        it('should add transaction to active transactions if admin', async () => {
            const mockUser = {
                updateOne: jest.fn().mockResolvedValue({})
            };
            User.findById.mockResolvedValue(mockUser);

            const res = await request(app)
                .put('/trans123/move-to-activetransactions')
                .send({ isAdmin: true, userId: 'user123' });

            expect(res.status).toBe(200);
            expect(res.body).toBe('Added to Active Transaction');
            expect(User.findById).toHaveBeenCalledWith('user123');
            expect(mockUser.updateOne).toHaveBeenCalledWith({
                $push: { activeTransactions: 'trans123' }
            });
        });

        it('should return 500 if updating user active transactions fails', async () => {
            User.findById.mockRejectedValue(new Error('Find error'));

            const res = await request(app)
                .put('/trans123/move-to-activetransactions')
                .send({ isAdmin: true, userId: 'user123' });

            expect(res.status).toBe(500);
        });
    });

    describe('PUT /:id/move-to-prevtransactions', () => {
        it('should return 403 if request is not made by admin', async () => {
            const res = await request(app)
                .put('/trans123/move-to-prevtransactions')
                .send({ isAdmin: false, userId: 'user123' });

            expect(res.status).toBe(403);
            expect(res.body).toBe('Only Admin can do this');
        });

        it('should move transaction to prev transactions if admin', async () => {
            const mockUser = {
                updateOne: jest.fn().mockResolvedValue({})
            };
            User.findById.mockResolvedValue(mockUser);

            const res = await request(app)
                .put('/trans123/move-to-prevtransactions')
                .send({ isAdmin: true, userId: 'user123' });

            expect(res.status).toBe(200);
            expect(res.body).toBe('Added to Prev transaction Transaction');
            expect(User.findById).toHaveBeenCalledWith('user123');
            expect(mockUser.updateOne).toHaveBeenNthCalledWith(1, {
                $pull: { activeTransactions: 'trans123' }
            });
            expect(mockUser.updateOne).toHaveBeenNthCalledWith(2, {
                $push: { prevTransactions: 'trans123' }
            });
        });

        it('should return 500 if updating prev transactions fails', async () => {
            User.findById.mockRejectedValue(new Error('User error'));

            const res = await request(app)
                .put('/trans123/move-to-prevtransactions')
                .send({ isAdmin: true, userId: 'user123' });

            expect(res.status).toBe(500);
        });
    });

    describe('DELETE /deleteuser/:id', () => {
        it('should return 403 if user is not authorized to delete account', async () => {
            const res = await request(app)
                .delete('/deleteuser/user123')
                .send({ userId: 'otherUser', isAdmin: false });

            expect(res.status).toBe(403);
            expect(res.body).toBe('You can delete only your account!');
        });

        it('should delete user if user deletes own account', async () => {
            User.findByIdAndDelete.mockResolvedValue({});

            const res = await request(app)
                .delete('/deleteuser/user123')
                .send({ userId: 'user123', isAdmin: false });

            expect(res.status).toBe(200);
            expect(res.body).toBe('Account has been deleted');
            expect(User.findByIdAndDelete).toHaveBeenCalledWith('user123');
        });

        it('should delete user if admin initiates deletion', async () => {
            User.findByIdAndDelete.mockResolvedValue({});

            const res = await request(app)
                .delete('/deleteuser/user123')
                .send({ userId: 'adminUser', isAdmin: true });

            expect(res.status).toBe(200);
            expect(res.body).toBe('Account has been deleted');
            expect(User.findByIdAndDelete).toHaveBeenCalledWith('user123');
        });

        it('should return 500 if DB deletion throws error', async () => {
            User.findByIdAndDelete.mockRejectedValue(new Error('Delete error'));

            const res = await request(app)
                .delete('/deleteuser/user123')
                .send({ userId: 'user123' });

            expect(res.status).toBe(500);
        });
    });
});
```
