    jest.restoreAllMocks();
  });

    test('returns success true and logs payload', async () => {
      const logSpy = jest.spyOn(console, 'log').mockImplementation(() => {});
      const result = await customerDataRequest(topic, shop, sampleBody);
      expect(result).toEqual({ success: true });
      expect(logSpy).toHaveBeenCalledWith(`Handle ${topic} for ${shop}`);
      expect(logSpy).toHaveBeenCalledWith(sampleBody);
    });

    test('catches error and returns success false', async () => {
      const error = new Error('log failure');
      jest.spyOn(console, 'log').mockImplementation(() => {
        throw error;
      });
      const errorSpy = jest.spyOn(console, 'error').mockImplementation(() => {});
      const result = await customerDataRequest(topic, shop, sampleBody);
      expect(result).toEqual({ success: false });
      expect(errorSpy).toHaveBeenCalledWith(error);
    });
  });
