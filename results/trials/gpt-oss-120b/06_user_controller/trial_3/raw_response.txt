```javascript
import {
  updateProfile,
  requestPasswordReset,
  resetPassword,
  User,
  EmailService,
  CryptoHelper
} from '../dataset/06_user_controller.js';

describe('updateProfile', () => {
  const makeRes = () => ({
    status: jest.fn().mockReturnThis(),
    json: jest.fn()
  });

  beforeEach(() => {
    jest.clearAllMocks();
  });

  test('returns 401 when no user identifier is provided', async () => {
    const req = { user: {}, params: {}, body: {} };
    const res = makeRes();

    await updateProfile(req, res);

    expect(res.status).toHaveBeenCalledWith(401);
    expect(res.json).toHaveBeenCalledWith({
      success: false,
      error: 'Authentication required'
    });
  });

  test('returns 404 when user not found', async () => {
    const req = { user: { id: 'uid' }, body: {} };
    const res = makeRes();

    jest.spyOn(User, 'findById').mockResolvedValueOnce(null);

    await updateProfile(req, res);

    expect(User.findById).toHaveBeenCalledWith('uid');
    expect(res.status).toHaveBeenCalledWith(404);
    expect(res.json).toHaveBeenCalledWith({
      success: false,
      error: 'User not found'
    });
  });

  test('returns 400 for empty name', async () => {
    const req = { user: { id: 'uid' }, body: { name: '   ' } };
    const res = makeRes();

    jest.spyOn(User, 'findById').mockResolvedValueOnce({ id: 'uid' });

    await updateProfile(req, res);

    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      success: false,
      error: 'Name cannot be empty'
    });
  });

  test('returns 400 for bio exceeding limit', async () => {
    const longBio = 'a'.repeat(251);
    const req = { user: { id: 'uid' }, body: { bio: longBio } };
    const res = makeRes();

    jest.spyOn(User, 'findById').mockResolvedValueOnce({ id: 'uid' });

    await updateProfile(req, res);

    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      success: false,
      error: 'Bio must be a string up to 250 characters'
    });
  });

  test('returns 400 for invalid phone format', async () => {
    const req = { user: { id: 'uid' }, body: { phone: '12345' } };
    const res = makeRes();

    jest.spyOn(User, 'findById').mockResolvedValueOnce({ id: 'uid' });

    await updateProfile(req, res);

    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      success: false,
      error: 'Invalid international phone number format'
    });
  });

  test('returns 400 for malformed avatarUrl', async () => {
    const req = { user: { id: 'uid' }, body: { avatarUrl: 'not-a-url' } };
    const res = makeRes();

    jest.spyOn(User, 'findById').mockResolvedValueOnce({ id: 'uid' });

    await updateProfile(req, res);

    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      success: false,
      error: 'Invalid avatar URL'
    });
  });

  test('successfully updates profile with valid fields', async () => {
    const req = {
      user: { id: 'uid' },
      body: {
        name: 'Alice',
        bio: 'Hello world',
        phone: '+12345678901',
        avatarUrl: 'https://example.com/avatar.png'
      }
    };
    const res = makeRes();

    jest.spyOn(User, 'findById').mockResolvedValueOnce({ id: 'uid' });
    const updateSpy = jest.spyOn(User, 'update').mockResolvedValueOnce({
      id: 'uid',
      name: 'Alice',
      bio: 'Hello world',
      phone: '+12345678901',
      avatarUrl: 'https://example.com/avatar.png'
    });

    await updateProfile(req, res);

    expect(updateSpy).toHaveBeenCalledWith('uid', {
      name: 'Alice',
      bio: 'Hello world',
      phone: '+12345678901',
      avatarUrl: 'https://example.com/avatar.png'
    });
    expect(res.status).toHaveBeenCalledWith(200);
    expect(res.json).toHaveBeenCalledWith({
      success: true,
      message: 'Profile updated successfully',
      data: expect.objectContaining({ id: 'uid' })
    });
  });
});

describe('requestPasswordReset', () => {
  const makeRes = () => ({
    status: jest.fn().mockReturnThis(),
    json: jest.fn()
  });

  beforeEach(() => {
    jest.clearAllMocks();
  });

  test('returns 400 when email is missing', async () => {
    const req = { body: {} };
    const res = makeRes();

    await requestPasswordReset(req, res);

    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      success: false,
      error: 'Email is required'
    });
  });

  test('returns 400 for invalid email format', async () => {
    const req = { body: { email: 'invalid-email' } };
    const res = makeRes();

    await requestPasswordReset(req, res);

    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      success: false,
      error: 'Invalid email format'
    });
  });

  test('responds with generic success when email not found', async () => {
    const req = { body: { email: 'nosuch@example.com' } };
    const res = makeRes();

    jest.spyOn(User, 'findByEmail').mockResolvedValueOnce(null);

    await requestPasswordReset(req, res);

    expect(User.findByEmail).toHaveBeenCalledWith('nosuch@example.com');
    expect(res.status).toHaveBeenCalledWith(200);
    expect(res.json).toHaveBeenCalledWith({
      success: true,
      message: 'If an account exists with that email, a password reset link has been dispatched'
    });
  });

  test('generates token, saves it, sends email, and returns generic success', async () => {
    const req = { body: { email: 'user@example.com' } };
    const res = makeRes();

    const fakeUser = { id: 'uid', email: 'user@example.com' };
    jest.spyOn(User, 'findByEmail').mockResolvedValueOnce(fakeUser);
    const tokenSpy = jest.spyOn(CryptoHelper, 'generateToken').mockReturnValue('generated-token');
    const saveSpy = jest.spyOn(User, 'saveResetToken').mockResolvedValueOnce(true);
    const emailSpy = jest.spyOn(EmailService, 'sendResetEmail').mockResolvedValueOnce(true);

    await requestPasswordReset(req, res);

    expect(tokenSpy).toHaveBeenCalled();
    expect(saveSpy).toHaveBeenCalledWith('uid', 'generated-token', expect.any(Date));
    expect(emailSpy).toHaveBeenCalledWith('user@example.com', 'generated-token');
    expect(res.status).toHaveBeenCalledWith(200);
    expect(res.json).toHaveBeenCalledWith({
      success: true,
      message: 'If an account exists with that email, a password reset link has been dispatched'
    });
  });
});

describe('resetPassword', () => {
  const makeRes = () => ({
    status: jest.fn().mockReturnThis(),
    json: jest.fn()
  });

  beforeEach(() => {
    jest.clearAllMocks();
  });

  test('returns 400 when token or newPassword missing', async () => {
    const req = { body: { token: 'abc' } };
    const res = makeRes();

    await resetPassword(req, res);

    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      success: false,
      error: 'Token and newPassword are required'
    });
  });

  test('returns 400 when newPassword too short', async () => {
    const req = { body: { token: 'tok', newPassword: 'short' } };
    const res = makeRes();

    await resetPassword(req, res);

    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      success: false,
      error: 'New password must be at least 8 characters long'
    });
  });

  test('returns 400 when token not associated with any user', async () => {
    const req = { body: { token: 'badtoken', newPassword: 'validPass123' } };
    const res = makeRes();

    jest.spyOn(User, 'findByResetToken').mockResolvedValueOnce(null);

    await resetPassword(req, res);

    expect(User.findByResetToken).toHaveBeenCalledWith('badtoken');
    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      success: false,
      error: 'Invalid or expired password reset token'
    });
  });

  test('returns 400 when token is expired', async () => {
    const req = { body: { token: 'expiring', newPassword: 'validPass123' } };
    const res = makeRes();

    const expiredUser = {
      id: 'uid',
      resetTokenExpires: new Date(Date.now() - 1000).toISOString()
    };
    jest.spyOn(User, 'findByResetToken').mockResolvedValueOnce(expiredUser);

    await resetPassword(req, res);

    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      success: false,
      error: 'Password reset token has expired'
    });
  });

  test('successfully resets password when token is valid', async () => {
    const req = { body: { token: 'validtoken', newPassword: 'newStrongPass' } };
    const res = makeRes();

    const user = {
      id: 'uid',
      resetTokenExpires: new Date(Date.now() + 60000).toISOString()
    };
    jest.spyOn(User, 'findByResetToken').mockResolvedValueOnce(user);
    const hashSpy = jest.spyOn(CryptoHelper, 'hashPassword').mockResolvedValueOnce('hashed-newStrongPass');
    const updateSpy = jest.spyOn(User, 'update').mockResolvedValueOnce({});

    await resetPassword(req, res);

    expect(hashSpy).toHaveBeenCalledWith('newStrongPass');
    expect(updateSpy).toHaveBeenCalledWith('uid', {
      password: 'hashed-newStrongPass',
      resetToken: null,
      resetTokenExpires: null
    });
    expect(res.status).toHaveBeenCalledWith(200);
    expect(res.json).toHaveBeenCalledWith({
      success: true,
      message: 'Password has been successfully reset'
    });
  });
});
```
