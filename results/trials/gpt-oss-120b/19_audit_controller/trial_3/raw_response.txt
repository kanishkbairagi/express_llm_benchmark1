```javascript
import { jest } from '@jest/globals';
import {
  AuditLogModel,
  getAuditLogs,
  createAuditEntry
} from '../dataset/19_audit_controller.js';

const mockRes = () => {
  const res = {};
  res.status = jest.fn().mockReturnValue(res);
  res.json = jest.fn().mockReturnValue(res);
  return res;
};

describe('getAuditLogs controller', () => {
  afterEach(() => {
    jest.restoreAllMocks();
  });

  test('denies access for non‑admin/auditor roles', async () => {
    const req = { user: { role: 'user' }, query: {} };
    const res = mockRes();

    await getAuditLogs(req, res);

    expect(res.status).toHaveBeenCalledWith(403);
    expect(res.json).toHaveBeenCalledWith({
      success: false,
      error: 'Access denied: Requires administrator or auditor privileges'
    });
  });

  test('returns 400 for invalid pagination parameters', async () => {
    const req = {
      user: { role: 'admin' },
      query: { page: '0', limit: '-5' }
    };
    const res = mockRes();

    await getAuditLogs(req, res);

    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      success: false,
      error: 'Pagination parameters must be positive integers (limit <= 100)'
    });
  });

  test('returns 400 for unknown action filter', async () => {
    const req = {
      user: { role: 'auditor' },
      query: { action: 'unknown' }
    };
    const res = mockRes();

    await getAuditLogs(req, res);

    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      success: false,
      error: expect.stringContaining('Invalid action filter')
    });
  });

  test('returns 400 for unknown severity filter', async () => {
    const req = {
      user: { role: 'admin' },
      query: { severity: 'fatal' }
    };
    const res = mockRes();

    await getAuditLogs(req, res);

    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      success: false,
      error: expect.stringContaining('Invalid severity filter')
    });
  });

  test('returns 400 for malformed startDate', async () => {
    const req = {
      user: { role: 'admin' },
      query: { startDate: 'not-a-date' }
    };
    const res = mockRes();

    await getAuditLogs(req, res);

    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      success: false,
      error: 'Invalid startDate format'
    });
  });

  test('returns 200 with correct pagination data on success', async () => {
    const mockFind = jest
      .spyOn(AuditLogModel, 'findWithFilters')
      .mockResolvedValue({
        total: 45,
        logs: [{ id: 'log1' }, { id: 'log2' }]
      });

    const req = {
      user: { role: 'auditor' },
      query: { page: '2', limit: '20', action: 'login', severity: 'Info' }
    };
    const res = mockRes();

    await getAuditLogs(req, res);

    expect(mockFind).toHaveBeenCalledWith(
      {
        action: 'LOGIN',
        severity: 'info'
      },
      {
        skip: 20,
        limit: 20
      }
    );
    expect(res.status).toHaveBeenCalledWith(200);
    expect(res.json).toHaveBeenCalledWith({
      success: true,
      data: {
        total: 45,
        page: 2,
        totalPages: 3,
        logs: [{ id: 'log1' }, { id: 'log2' }]
      }
    });
  });

  test('returns totalPages = 1 when total is zero', async () => {
    jest.spyOn(AuditLogModel, 'findWithFilters').mockResolvedValue({
      total: 0,
      logs: []
    });

    const req = {
      user: { role: 'admin' },
      query: {}
    };
    const res = mockRes();

    await getAuditLogs(req, res);

    expect(res.json).toHaveBeenCalledWith({
      success: true,
      data: {
        total: 0,
        page: 1,
        totalPages: 1,
        logs: []
      }
    });
  });
});

describe('createAuditEntry controller', () => {
  afterEach(() => {
    jest.restoreAllMocks();
  });

  test('returns 400 when required fields are missing', async () => {
    const req = {
      user: { id: 'u123' },
      body: { action: 'CREATE' } // missing targetResourceId and targetResourceType
    };
    const res = mockRes();

    await createAuditEntry(req, res);

    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      success: false,
      error: 'action, targetResourceId, and targetResourceType are required'
    });
  });

  test('returns 400 for invalid action value', async () => {
    const req = {
      user: { id: 'u123' },
      body: {
        action: 'invalid',
        targetResourceId: 'r1',
        targetResourceType: 'file'
      }
    };
    const res = mockRes();

    await createAuditEntry(req, res);

    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      success: false,
      error: expect.stringContaining('Invalid action')
    });
  });

  test('returns 400 for invalid severity level', async () => {
    const req = {
      user: { id: 'u123' },
      body: {
        action: 'delete',
        targetResourceId: 'r1',
        targetResourceType: 'file',
        severity: 'urgent'
      }
    };
    const res = mockRes();

    await createAuditEntry(req, res);

    expect(res.status).toHaveBeenCalledWith(400);
    expect(res.json).toHaveBeenCalledWith({
      success: false,
      error: expect.stringContaining('Invalid severity level')
    });
  });

  test('creates audit entry and returns 201 on success', async () => {
    const mockCreate = jest
      .spyOn(AuditLogModel, 'create')
      .mockImplementation(async (data) => ({
        id: 'audit_123',
        ...data,
        timestamp: new Date('2023-01-01T00:00:00Z')
      }));

    const req = {
      user: { id: 'user42' },
      ip: '10.0.0.5',
      body: {
        action: 'update',
        targetResourceId: 'res99',
        targetResourceType: 'document',
        details: { field: 'name' },
        severity: 'Warning'
      }
    };
    const res = mockRes();

    await createAuditEntry(req, res);

    expect(mockCreate).toHaveBeenCalledWith({
      actorId: 'user42',
      action: 'UPDATE',
      targetResourceId: 'res99',
      targetResourceType: 'document',
      severity: 'warning',
      details: { field: 'name' },
      ipAddress: '10.0.0.5'
    });
    expect(res.status).toHaveBeenCalledWith(201);
    expect(res.json).toHaveBeenCalledWith({
      success: true,
      data: expect.objectContaining({
        id: 'audit_123',
        actorId: 'user42',
        action: 'UPDATE',
        targetResourceId: 'res99',
        targetResourceType: 'document',
        severity: 'warning',
        details: { field: 'name' },
        ipAddress: '10.0.0.5'
      })
    });
  });

  test('defaults actorId to "system" when user missing', async () => {
    const mockCreate = jest.spyOn(AuditLogModel, 'create').mockResolvedValue({ id: 'audit_x' });

    const req = {
      // no user object
      headers: { 'x-forwarded-for': '203.0.113.7' },
      body: {
        action: 'login',
        targetResourceId: 'u1',
        targetResourceType: 'user'
      }
    };
    const res = mockRes();

    await createAuditEntry(req, res);

    expect(mockCreate).toHaveBeenCalledWith(
      expect.objectContaining({ actorId: 'system', ipAddress: '203.0.113.7' })
    );
    expect(res.status).toHaveBeenCalledWith(201);
  });
});
```
