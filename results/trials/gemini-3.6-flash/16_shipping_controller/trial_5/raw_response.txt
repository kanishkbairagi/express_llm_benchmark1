```javascript
import {
  validateShippingAddress,
  calculateShippingRates,
  AddressValidationService,
  CarrierRateService
} from '../dataset/16_shipping_controller.js';

const mockReq = (body) => ({ body });

const mockRes = () => {
  const res = {};
  res.status = jest.fn().mockReturnValue(res);
  res.json = jest.fn().mockReturnValue(res);
  return res;
};

describe('16_shipping_controller unit tests', () => {
  afterEach(() => {
    jest.restoreAllMocks();
  });

  describe('validateShippingAddress', () => {
    it('should return 400 if required address fields are missing', async () => {
      const req = mockReq({ street: '123 Main St', city: 'Anytown' }); // state and postalCode missing
      const res = mockRes();

      await validateShippingAddress(req, res);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.json).toHaveBeenCalledWith({
        success: false,
        error: 'Incomplete address: street, city, state, and postalCode are required'
      });
    });

    it('should return 400 if req.body is undefined', async () => {
      const req = {};
      const res = mockRes();

      await validateShippingAddress(req, res);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.json).toHaveBeenCalledWith({
        success: false,
        error: 'Incomplete address: street, city, state, and postalCode are required'
      });
    });

    it('should return 422 for restricted destination countries', async () => {
      const req = mockReq({
        street: '123 Main St',
        city: 'Havana',
        state: 'Havana',
        postalCode: '10100',
        country: 'CU'
      });
      const res = mockRes();

      await validateShippingAddress(req, res);

      expect(res.status).toHaveBeenCalledWith(422);
      expect(res.json).toHaveBeenCalledWith({
        success: false,
        error: 'Shipping is not available to destination country code "CU"'
      });
    });

    it('should return 400 for an invalid US postal code format', async () => {
      const req = mockReq({
        street: '123 Main St',
        city: 'Springfield',
        state: 'IL',
        postalCode: 'ABCDE',
        country: 'US'
      });
      const res = mockRes();

      await validateShippingAddress(req, res);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.json).toHaveBeenCalledWith({
        success: false,
        error: 'Invalid US postal code format (expected 12345 or 12345-6789)'
      });
    });

    it('should return 200 and verify address for a valid US address (default country)', async () => {
      const req = mockReq({
        street: ' 123 Main St ',
        city: ' Springfield ',
        state: ' IL ',
        postalCode: ' 62701 '
      });
      const res = mockRes();

      await validateShippingAddress(req, res);

      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.json).toHaveBeenCalledWith({
        success: true,
        message: 'Address verified',
        data: {
          isValid: true,
          normalized: {
            street: '123 Main St',
            city: 'Springfield',
            state: 'IL',
            postalCode: '62701',
            country: 'US'
          }
        }
      });
    });

    it('should return 200 for valid US ZIP+4 format and non-US valid address', async () => {
      const req = mockReq({
        street: '456 Queen St',
        city: 'Toronto',
        state: 'ON',
        postalCode: 'M5V 2T6',
        country: 'ca'
      });
      const res = mockRes();

      await validateShippingAddress(req, res);

      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.json).toHaveBeenCalledWith({
        success: true,
        message: 'Address verified',
        data: {
          isValid: true,
          normalized: {
            street: '456 Queen St',
            city: 'Toronto',
            state: 'ON',
            postalCode: 'M5V 2T6',
            country: 'CA'
          }
        }
      });
    });

    it('should return 500 if AddressValidationService throws an error', async () => {
      jest.spyOn(AddressValidationService, 'verify').mockRejectedValueOnce(new Error('Service unavailable'));

      const req = mockReq({
        street: '123 Main St',
        city: 'Springfield',
        state: 'IL',
        postalCode: '62701',
        country: 'US'
      });
      const res = mockRes();

      await validateShippingAddress(req, res);

      expect(res.status).toHaveBeenCalledWith(500);
      expect(res.json).toHaveBeenCalledWith({
        success: false,
        error: 'Address validation failed',
        details: 'Service unavailable'
      });
    });
  });

  describe('calculateShippingRates', () => {
    it('should return 400 if destination or destination fields are missing', async () => {
      const req = mockReq({ destination: { country: 'US' } }); // postalCode missing
      const res = mockRes();

      await calculateShippingRates(req, res);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.json).toHaveBeenCalledWith({
        success: false,
        error: 'Destination country and postalCode are required'
      });
    });

    it('should return 422 if destination country is restricted', async () => {
      const req = mockReq({
        destination: { country: 'IR', postalCode: '12345' }
      });
      const res = mockRes();

      await calculateShippingRates(req, res);

      expect(res.status).toHaveBeenCalledWith(422);
      expect(res.json).toHaveBeenCalledWith({
        success: false,
        error: 'Carrier does not ship to "IR"'
      });
    });

    it('should return 400 if packageDetails is missing or not an object', async () => {
      const req = mockReq({
        destination: { country: 'US', postalCode: '90210' },
        packageDetails: 'not-an-object'
      });
      const res = mockRes();

      await calculateShippingRates(req, res);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.json).toHaveBeenCalledWith({
        success: false,
        error: 'packageDetails object is required'
      });
    });

    it('should return 400 if weightKg is not a positive number', async () => {
      const req = mockReq({
        destination: { country: 'US', postalCode: '90210' },
        packageDetails: { weightKg: 0, dimensions: { length: 10, width: 10, height: 10 } }
      });
      const res = mockRes();

      await calculateShippingRates(req, res);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.json).toHaveBeenCalledWith({
        success: false,
        error: 'Package weightKg must be a positive number'
      });
    });

    it('should return 400 if dimensions are invalid or non-positive', async () => {
      const req = mockReq({
        destination: { country: 'US', postalCode: '90210' },
        packageDetails: { weightKg: 2, dimensions: { length: 10, width: -5, height: 10 } }
      });
      const res = mockRes();

      await calculateShippingRates(req, res);

      expect(res.status).toHaveBeenCalledWith(400);
      expect(res.json).toHaveBeenCalledWith({
        success: false,
        error: 'Valid dimensions (length, width, height > 0) in cm are required'
      });
    });

    it('should calculate rates using actual weight when weightKg > volumetricWeight', async () => {
      // Dimensions 10x10x10 / 5000 = 0.2 kg volumetric weight. Weight 5kg.
      const req = mockReq({
        destination: { country: 'US', postalCode: '90210' },
        packageDetails: {
          weightKg: 5,
          dimensions: { length: 10, width: 10, height: 10 }
        }
      });
      const res = mockRes();

      await calculateShippingRates(req, res);

      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.json).toHaveBeenCalledWith({
        success: true,
        data: {
          billableWeightKg: 5,
          isVolumetric: false,
          options: [
            { service: 'standard', cost: 17.5, currency: 'USD', estimatedDelivery: '3-5 business days' },
            { service: 'express', cost: 37, currency: 'USD', estimatedDelivery: '1-2 business days' },
            { service: 'overnight', cost: 70, currency: 'USD', estimatedDelivery: 'Next business day' }
          ]
        }
      });
    });

    it('should calculate rates using volumetric weight when volumetricWeight > weightKg', async () => {
      // Dimensions 50x50x50 / 5000 = 25 kg volumetric weight. Actual weight 2kg.
      const req = mockReq({
        destination: { country: 'CA', postalCode: 'M5V 2T6' },
        packageDetails: {
          weightKg: 2,
          dimensions: { length: 50, width: 50, height: 50 }
        }
      });
      const res = mockRes();

      await calculateShippingRates(req, res);

      expect(res.status).toHaveBeenCalledWith(200);
      expect(res.json).toHaveBeenCalledWith({
        success: true,
        data: {
          billableWeightKg: 25,
          isVolumetric: true,
          options: [
            { service: 'standard', cost: 67.5, currency: 'USD', estimatedDelivery: '3-5 business days' },
            { service: 'express', cost: 137, currency: 'USD', estimatedDelivery: '1-2 business days' },
            { service: 'overnight', cost: 250, currency: 'USD', estimatedDelivery: 'Next business day' }
          ]
        }
      });
    });

    it('should return 500 if CarrierRateService throws an error', async () => {
      jest.spyOn(CarrierRateService, 'getRates').mockRejectedValueOnce(new Error('Carrier API error'));

      const req = mockReq({
        destination: { country: 'US', postalCode: '90210' },
        packageDetails: {
          weightKg: 1,
          dimensions: { length: 10, width: 10, height: 10 }
        }
      });
      const res = mockRes();

      await calculateShippingRates(req, res);

      expect(res.status).toHaveBeenCalledWith(500);
      expect(res.json).toHaveBeenCalledWith({
        success: false,
        error: 'Failed to compute shipping rates',
        details: 'Carrier API error'
      });
    });
  });
});
```
